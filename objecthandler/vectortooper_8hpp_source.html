46 template <class T>
47 void vectorToOper(T begin, T end, OPER &xVector) {
48 std::size_t size = end - begin;
49 if (size == 0) {
50 xVector.xltype = xltypeErr;
51 xVector.val.err = xlerrNA;
52 return;
53 }
54
55 if (FunctionCall::instance().callerDimensions() == CallerDimensions::Row) {
56 xVector.val.array.columns = size;
57 xVector.val.array.rows = 1;
58 } else {
59 xVector.val.array.rows = size;
60 xVector.val.array.columns = 1;
61 }
62
63 xVector.val.array.lparray = new OPER[size];
64 xVector.xltype = xltypeMulti | xlbitDLLFree;
65 for (unsigned int i=0; i<size; ++i, ++begin)
66 scalarToOper(*begin, xVector.val.array.lparray[i], false);
67 }

69}
