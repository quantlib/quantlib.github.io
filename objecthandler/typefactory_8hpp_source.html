65 template<typename KeyClass>
66 void *getType(const KeyClass& id) {
67 typename RegistryClass::TypeMapPtr typeMap = getTypeMap();
68 KeyClass idUpper = uppercase(id);
69 typename RegistryClass::TypeMap::iterator i;
70 for (i = typeMap->begin(); i != typeMap->end(); ++i)
71 if (uppercase(i->first) == idUpper)
72 return i->second;
73 OH_FAIL("Unknown id for Type: " << id);
74 }

77 bool checkType(const std::string& id) {
78 typename RegistryClass::TypeMapPtr type_map;
79 typename RegistryClass::AllTypeMap::const_iterator i =
80 RegistryClass::instance().getAllTypesMap().find(typeid(T).name());
81 if (i == RegistryClass::instance().getAllTypesMap().end()) {
82 return false;
83 } else {
84 type_map = i->second;
85 }
86 std::string idUpper = boost::algorithm::to_upper_copy(id);
87 for (typename RegistryClass::TypeMap::iterator i = type_map->begin(); i != type_map->end(); ++i)
88 if (boost::algorithm::to_upper_copy(i->first) == idUpper)
89 return true;
90 return false;
91 }

102 void unregisterTypes() {
103 typename RegistryClass::TypeMapPtr typeMap = getTypeMap();
104 typename RegistryClass::TypeMap::iterator i;
105 for (i = typeMap->begin(); i != typeMap->end(); ++i)
106 delete static_cast<T*>(i->second);
107 RegistryClass::instance().deleteTypeMap(typeid(T).name());
108 }
110
111 protected:
112 const typename RegistryClass::TypeMapPtr &getTypeMap() {
113 typename RegistryClass::AllTypeMap::const_iterator i =
114 RegistryClass::instance().getAllTypesMap().find(typeid(T).name());
115 OH_REQUIRE(i != RegistryClass::instance().getAllTypesMap().end(),
116 "Error retrieving Enumeration from Registry - the type '"
117 << typeid(T).name() << "' is not available!");
118 return i->second;
119 }

147
148 try {
149 std::vector<T> returnValue;

151 returnValue.push_back(Create<T>()(*i));
152 return returnValue;
153 } catch (const std::exception &e) {
154 OH_FAIL("vectorStringToEnum: error converting parameter '" << paramName
