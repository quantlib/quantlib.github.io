93 template <typename KeyClass>
94 inline void Registry<KeyClass>::registerType(
95 const std::string &mapID,
96 const KeyClass &typeID,
97 void *type) const {
98 AllTypeMap &allTypeMap = getAllTypesMap();
99 TypeMapPtr typeMapPtr;
100 typename AllTypeMap::const_iterator i = allTypeMap.find(mapID);
101 if (i == allTypeMap.end()) {
102 typeMapPtr = TypeMapPtr(new TypeMap);
103 allTypeMap[mapID] = typeMapPtr;
104 } else {
105 typeMapPtr = i->second;
106 }
107 (*typeMapPtr)[typeID] = type;
108 }

119 template <typename KeyClass>
120 inline std::vector<std::string> Registry<KeyClass>::getTypeElements(
121 const std::string& id) const {
122
123 AllTypeMap &allTypeMap = getAllTypesMap();
124 typename AllTypeMap::const_iterator map = allTypeMap.find(id);
125 OH_REQUIRE(map != allTypeMap.end(), "Registry::getTypeElements: invalid enum id: " + id);
126 std::vector<std::string> ret;
127 for(typename TypeMap::const_iterator i = map->second->begin(); i != map->second->end(); ++i)
128 ret.push_back(i->first);
129 return ret;
130 }

132 template <>
133 inline std::vector<std::string> Registry<KeyPair>::getTypeElements(
134 const std::string& id) const {
135
136 AllTypeMap &allTypeMap = getAllTypesMap();
137 AllTypeMap::const_iterator map = allTypeMap.find(id);
138 OH_REQUIRE(map != allTypeMap.end(), "Registry::getTypeElements: invalid enum id: " + id);
139 std::vector<std::string> ret;
140 for(TypeMap::const_iterator i = map->second->begin(); i != map->second->end(); ++i) {
141 std::ostringstream s;
142 s << i->first.first << " | " << i->first.second;
143 ret.push_back(s.str());
144 }
145 return ret;
146 }
