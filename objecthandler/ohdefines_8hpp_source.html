87#define OH_GET_OBJECT( NAME, ID, OBJECT_CLASS ) \
88 boost::shared_ptr<OBJECT_CLASS > NAME; \
89 ObjectHandler::Repository::instance().retrieveObject(NAME, ID);

92#define OH_GET_OBJECT_DEFAULT( NAME, ID, OBJECT_CLASS ) \
93 boost::shared_ptr<OBJECT_CLASS > NAME; \
94 if (!ID.empty()) { \
95 ObjectHandler::Repository::instance().retrieveObject(NAME, ID); \
96 }

99#define OH_GET_REFERENCE( NAME, ID, OBJECT_CLASS, LIBRARY_CLASS ) \
100 OH_GET_OBJECT(NAME ## temp, ID, OBJECT_CLASS ) \
101 boost::shared_ptr<LIBRARY_CLASS> NAME; \
102 NAME ## temp->getLibraryObject(NAME);
103
105#define OH_GET_REFERENCE_DEFAULT( NAME, ID, OBJECT_CLASS, LIBRARY_CLASS ) \
106 boost::shared_ptr<LIBRARY_CLASS> NAME; \
107 if (!ID.empty()) { \
108 OH_GET_OBJECT(NAME ## temp, ID, OBJECT_CLASS ) \
109 NAME ## temp->getLibraryObject(NAME); \
110 }
