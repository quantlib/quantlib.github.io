175 inline std::set<std::string> ValueObject::getPropertyNames() const {

177 for(std::map<std::string, property_t>::const_iterator i

183 inline property_t ValueObject::getProperty(const std::string& name) const {
184 if(userProperties.find(name) != userProperties.end())
185 return userProperties.find(name)->second;
186 else
187 return getSystemProperty(name);
188 }
189
190 inline bool ValueObject::hasProperty(const std::string& name) const {
191 return userProperties.find(name) != userProperties.end();
192 }
193
194 inline void ValueObject::setProperty(const std::string& name, const property_t& value) {
195 const std::set<std::string>& sysNames = getSystemPropertyNames();
196 if(sysNames.find(name) != sysNames.end())
197 setSystemProperty(name, value);
198 else
199 userProperties[name] = value;
200 }
