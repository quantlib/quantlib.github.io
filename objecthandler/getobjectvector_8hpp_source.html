49 template <class ObjectClass>
50 std::vector<boost::shared_ptr<ObjectClass> > getObjectVector(
51 const std::vector<std::string> &objectIDs,
52 const int &nestingLevel = 0,
53 bool includeGroups = false) {
54
55 OH_REQUIRE(nestingLevel < 10, "getObjectVector - nesting level exceeds 10. "
56 "(Possible infinite recursion?)");
57
58 std::vector<boost::shared_ptr<ObjectClass> > ret;
59
60 for (std::vector<std::string>::const_iterator i = objectIDs.begin();
61 i != objectIDs.end(); ++i) {
62
63 std::string objectId = *i;
64
65 // If the object ID is a null string then silently ignore.
66 // This will be revised to raise an exception in the absence of explicit
67 // indication from the user to do otherwise.
68 if (objectId.empty())
69 continue;
70
71 boost::shared_ptr<Object> object;
72 Repository::instance().retrieveObject(object, objectId);
73
74 boost::shared_ptr<Group> group =
75 boost::dynamic_pointer_cast<Group>(object);
76 if (group) {
77 std::vector<boost::shared_ptr<ObjectClass> > ret2 =
78 getObjectVector<ObjectClass>(group->list(), nestingLevel + 1, includeGroups);
79 ret.insert(ret.end(), ret2.begin(), ret2.end());
80 if(includeGroups) {
81 boost::shared_ptr<ObjectClass> objectDerived =
82 boost::dynamic_pointer_cast<ObjectClass>(object);
83 OH_REQUIRE(objectDerived, "Error converting Group with id '"
84 << objectId << "' - unable to convert to type '"
85 << typeid(ObjectClass).name() << "'");
86 ret.push_back(objectDerived);
87 }
88 } else {
89 boost::shared_ptr<ObjectClass> objectDerived =
90 boost::dynamic_pointer_cast<ObjectClass>(object);
91 OH_REQUIRE(objectDerived, "Error retrieving object with id '"
92 << objectId << "' - unable to convert reference to type '"
93 << typeid(ObjectClass).name() << "'");
94 ret.push_back(objectDerived);
95 }
96
97 }
98 return ret;
99 }
